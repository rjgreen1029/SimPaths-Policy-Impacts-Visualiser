// parseWorker.js — runs in a Web Worker, processes a batch of runs and posts results back.
// The main thread sends: { runs: [{ personText, benefitText, scenarioName, runId }] }
// The worker posts back: { metrics: [...] } or { error: string }

import * as d3 from "d3";
import {
  COLUMN_MAP, STRATIFIERS, STRAT_ONLY, ALL_DISPLAY_VARS, NUMERIC_VARS,
  REGION_MAP, DISABILITY_MAP, FINANCIAL_MAP, SOCIAL_CARE_MAP, PROV_SOCIAL_CARE_MAP,
  binAge, binChildren, aggregateSingleRun,
} from "./parseCore.js";

self.onmessage = async ({ data }) => {
  try {
    const allMetrics = [];
    for (const { personText, benefitText, scenarioName, runId } of data.runs) {
      const metrics = processRunTexts(personText, benefitText, scenarioName, runId);
      allMetrics.push(...metrics);
    }
    self.postMessage({ metrics: allMetrics });
  } catch (err) {
    self.postMessage({ error: err.message });
  }
};

function processRunTexts(personText, benefitText, scenarioName, runId) {
  const personRows  = d3.csvParse(personText);
  const benefitRows = d3.csvParse(benefitText);

  const benefitMap = new Map();
  for (const b of benefitRows) {
    const yr   = b.time || b.Time || b.Year;
    const buId = b.id_BenefitUnit || b.idbu || b.idBu;
    if (yr && buId) benefitMap.set(`${yr}_${buId}`, b);
  }

  const headers  = personRows.length ? Object.keys(personRows[0]) : [];
  const colIndex = [];
  const stagingCols = [];
  for (const h of headers) {
    const display = COLUMN_MAP[h];
    if (!display) continue;
    if (display.startsWith("_")) stagingCols.push([h, display]);
    else colIndex.push([h, display]);
  }

  const runRows = [];
  for (const p of personRows) {
    const yr   = p.time || p.Time || p.Year;
    const buId = p.idBu || p.idbu || p.id_BenefitUnit;
    const bRow = benefitMap.get(`${yr}_${buId}`) || {};
    const merged = { ...p, ...bRow };

    let wgt = +(merged.wgt || merged.Wgt || 1.0);
    if (isNaN(wgt) || wgt<=0) wgt=1.0;

    const renamed = { Year: +yr, wgt };

    for (let j=0; j<colIndex.length; j++) {
      const val = merged[colIndex[j][0]];
      if (val!==undefined) renamed[colIndex[j][1]] = val;
    }

    // Benefits Received flag
    let nonUC, uc;
    for (const [rawCol, stagingKey] of stagingCols) {
      if (stagingKey==="_NonUCFlag") nonUC = merged[rawCol];
      if (stagingKey==="_UCFlag")    uc    = merged[rawCol];
    }
    if (nonUC!==undefined || uc!==undefined) {
      const a  = String(nonUC??"").toLowerCase();
      const b_ = String(uc   ??"").toLowerCase();
      renamed["Benefits Received"] = (a==="true"||a==="1"||b_==="true"||b_==="1")
        ? "Benefits Received" : "No Benefits Received";
    }

    if (renamed["Age"]!=null)               renamed["Age"]               = binAge(renamed["Age"]);
    if (renamed["Number of children"]!=null) renamed["Number of children"]= binChildren(renamed["Number of children"]);
    if (renamed["Region"]!=null)             renamed["Region"]             = REGION_MAP[String(renamed["Region"])] ?? renamed["Region"];
    if (renamed["Disability Status"]!=null)  renamed["Disability Status"]  = DISABILITY_MAP[String(renamed["Disability Status"]).toLowerCase()]  ?? renamed["Disability Status"];
    if (renamed["Financial distress flag"]!=null) renamed["Financial distress flag"] = FINANCIAL_MAP[String(renamed["Financial distress flag"]).toLowerCase()] ?? renamed["Financial distress flag"];
    if (renamed["Need of social care"]!=null)     renamed["Need of social care"]     = SOCIAL_CARE_MAP[String(renamed["Need of social care"]).toLowerCase()]     ?? renamed["Need of social care"];
    if (renamed["Provided social care"]!=null)    renamed["Provided social care"]    = PROV_SOCIAL_CARE_MAP[String(renamed["Provided social care"]).toLowerCase()] ?? renamed["Provided social care"];

    // Gender recode
    if (renamed["Gender"]!=null) {
      const g = String(renamed["Gender"]).toLowerCase();
      renamed["Gender"] = (g==="1"||g==="true"||g==="male") ? "Male" : "Female";
    }

    runRows.push(renamed);
  }

  if (!runRows.length) return [];
  return aggregateSingleRun(runRows, scenarioName, runId);
}
